module main

use std.http

reason classify_ticket(text: string) -> string {
    goal: "Classify a customer support ticket into a category"
    constraints: [
        "Return one of: billing, technical, general, urgent",
        "If uncertain, prefer general",
        "Tickets mentioning outage or down are always urgent"
    ]
    examples: [
        ("My card was charged twice", "billing"),
        ("Server is down", "urgent"),
        ("How do I reset my password", "technical"),
        ("What integrations do you support", "general")
    ]
    fallback: "general"
}

evolve classify_ticket {
    track: true
    retrain_every: 500
    min_accuracy: 0.95
    approve: true
}

fn main() {
    server := http.new(port: 8080)

    server.post("/classify") => fn(req) -> Response {
        result := classify_ticket(req.body)
        Response.json({ category: result })
    }

    server.start()
}

The reason block

Declare what you want. The compiler figures out how.

You write 
reason classify_ticket(text: string) -> string {
    goal: "Classify support tickets"
    constraints: [
        "Return: billing, technical, general, urgent",
        "Outage or down → always urgent"
    ]
    examples: [
        ("Card charged twice", "billing"),
        ("Server is down", "urgent")
    ]
    fallback: "general"
}
Compiler generates 
// Optimized keyword-matching function
fn classify_ticket(text: &str) -> String {
    let lower = text.to_lowercase();

    // From constraints
    if lower.contains("outage")
       || lower.contains("down") {
        return "urgent".into();
    }

    // From examples → keyword extraction
    if lower.contains("card")
       || lower.contains("charged") {
        return "billing".into();
    }

    "general".into() // fallback
}
Today: smart keyword matching extracted from your examples and constraints.
Tomorrow: local AI model (Cortex V1). Your code stays exactly the same.
